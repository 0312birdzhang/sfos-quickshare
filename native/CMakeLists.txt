cmake_minimum_required(VERSION 3.10)
project(sfos_quickshare_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL 1.1.1 REQUIRED)
find_package(Threads REQUIRED)
add_library(quickshare-core STATIC src/core.cpp)
target_include_directories(quickshare-core PUBLIC include)
target_link_libraries(quickshare-core PUBLIC OpenSSL::Crypto)
# The EC key APIs are shared with Sailfish's OpenSSL 1.1.1; also build on 3.x.
target_compile_definitions(quickshare-core PRIVATE OPENSSL_SUPPRESS_DEPRECATED)
target_compile_options(quickshare-core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(sfos-quickshare-native-probe src/probe.cpp)
target_link_libraries(sfos-quickshare-native-probe PRIVATE quickshare-core)
add_executable(sfos-quickshare-native src/worker.cpp src/json.cpp)
target_link_libraries(sfos-quickshare-native PRIVATE quickshare-core Threads::Threads)
target_compile_options(sfos-quickshare-native PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_executable(quickshare-core-tests tests/core_tests.cpp)
target_link_libraries(quickshare-core-tests PRIVATE quickshare-core)
add_test(NAME native-protocol-core COMMAND quickshare-core-tests)
add_executable(quickshare-worker-tests tests/worker_tests.cpp src/json.cpp)
target_link_libraries(quickshare-worker-tests PRIVATE quickshare-core Threads::Threads)
add_test(NAME native-worker-safety COMMAND quickshare-worker-tests)
